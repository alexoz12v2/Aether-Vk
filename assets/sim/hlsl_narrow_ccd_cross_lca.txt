[numthreads(64, 1, 1)]


layout(local_size_x = 256) in;

struct PushConstants_narrow_ccd_cross_lca {
    RigidBodyArray scene_entities;
    SparseCollisions output_list;
    CrossSparseCollisions cross_output_list;
    ParticleData particles;
    PairBuffer pair_buffer;
    CrossPairBuffer cross_pair_buffer;
    LcaEntityArray lca_entities;
    float dt;
    float particle_radius;
    uint space_type;
};

[[vk::push_constant]]
PushConstants_narrow_ccd_cross_lca pc;

void narrow_ccd_cross_lca(uint3 DispatchThreadID : SV_DispatchThreadID) {
    uint pair_idx = DispatchThreadID.x;
    
    CrossPairBuffer cross_pairs = pc.cross_pair_buffer;
    if (pair_idx >= cross_pairs.count) return;
    
    CrossPair pair = cross_pairs.pairs[pair_idx];
    uint idA = pair.macro_id;
    uint idB = pair.micro_id;
    uint lca_id = pair.lca_id;

    RigidBody ent_A = pc.scene_entities.bodies[idA];
    RigidBody ent_B = pc.scene_entities.bodies[idB];
    
    uint shape_A = ent_A.shape_type;
    float3 extents_A = ent_A.shape_extents;
    float4 orient_A = ent_A.orientation;
    float3 pos_A = ent_A.position_mass.xyz;
    float3 vel_A = ent_A.linear_vel_drag.xyz;
    
    uint shape_B = ent_B.shape_type;
    float3 extents_B = ent_B.shape_extents;
    float4 orient_B = ent_B.orientation;
    float3 pos_B = ent_B.position_mass.xyz;
    float3 vel_B = ent_B.linear_vel_drag.xyz;

    LcaEntity lca = pc.lca_entities.entities[lca_id];
    
    float3 macro_rel_vel_au = vel_A - lca.center_vel;
    float3 macro_rel_pos_au = pos_A - lca.center_pos;

    pos_A = (macro_rel_pos_au / lca.scale) * AU_TO_KM;
    vel_A = (macro_rel_vel_au / lca.scale) * AU_TO_KM;
    extents_A = (extents_A / lca.scale) * AU_TO_KM;

    float3x3 rotA = quat_to_float3x3(orient_A);
    float4x4 trans_A = float4x4(float4(rotA[0], 0.0), float4(rotA[1], 0.0), float4(rotA[2], 0.0), float4(pos_A, 1.0));
    
    float3x3 rotB = quat_to_float3x3(orient_B);
    float4x4 trans_B = float4x4(float4(rotB[0], 0.0), float4(rotB[1], 0.0), float4(rotB[2], 0.0), float4(pos_B, 1.0));

    float toi, depth;
    float3 normal, contact;
    
#ifdef DEBUG_SHADERS
    debugPrintfEXT("Narrow Phase Cross CCD [pair %d]: idA=%d, idB=%d, lca_id=%d\n", pair_idx, idA, idB, lca_id);
#endif

    if (compute_toi_generic(shape_A, extents_A, trans_A, vel_A, shape_B, extents_B, trans_B, vel_B, 1e-3, 10, toi, normal, contact, depth)) {
        CrossSparseCollisions cross_out = pc.cross_output_list;
        uint count = atomicAdd(cross_out.count, 1);
        if (count < 4000u) {
            cross_out.pairs[count].valid = 1u;
            cross_out.pairs[count].macro_id = idA;
            cross_out.pairs[count].micro_id = idB;
            cross_out.pairs[count].lca_id = lca_id;
            cross_out.pairs[count].toi = toi;
            cross_out.pairs[count].contact_normal = float4(normal, 0.0);
            cross_out.pairs[count].contact_point = float4(contact, 1.0);
            cross_out.pairs[count].penetration_depth = depth;
        }
    }
}
